// ===================================================================== // westend-brain: the middleman between your page and three services. // POST / -> asks Claude (uses ANTHROPIC_API_KEY) // GET /live -> live pub details from Google (uses GOOGLE_PLACES_KEY) // + today's fixtures from TheSportsDB (free key, no secret) // Your keys live in Cloudflare as secrets ("env"), never in this code. // ===================================================================== // The only website allowed to use this Worker: your GitHub Pages address const ALLOWED_ORIGIN = "https://laurien101712.github.io"; // A box around Galway's West End (south-west corner and north-east corner) const WEST_END_BOX = { low: { latitude: 53.2688, longitude: -9.0620 }, high: { latitude: 53.2722, longitude: -9.0555 } }; // The Google fields we ask for. Fewer fields = cheaper, so only what the page uses. const GOOGLE_FIELDS = [ "places.id", "places.displayName", "places.location", "places.rating", "places.userRatingCount", "places.currentOpeningHours.openNow", "places.allowsDogs", "places.liveMusic", "places.goodForWatchingSports", "places.servesLunch", "places.servesDinner", "places.googleMapsUri" ].join(","); // TheSportsDB leagues to check for today's fixtures (league ID: label) const LEAGUES = { 4328: "Premier League", 5564: "All-Ireland Football", 5565: "All-Ireland Hurling", 5577: "National Football League", 5578: "National Hurling League" }; // Rugby is checked by sport, keeping only matches involving these teams const RUGBY_TEAMS = /ireland|connacht|munster|leinster|ulster/i; export default { async fetch(request, env) { const cors = { "Access-Control-Allow-Origin": ALLOWED_ORIGIN, "Access-Control-Allow-Methods": "GET, POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type" }; // Browser's "may I?" check before the real request if (request.method === "OPTIONS") return reply(null, 204, cors); // Only your page may use this Worker if (request.headers.get("Origin") !== ALLOWED_ORIGIN) return reply({ error: "Not allowed" }, 403, cors); const path = new URL(request.url).pathname; if (request.method === "GET" && path === "/live") return reply(await getLive(env), 200, cors); if (request.method === "POST" && (path === "/" || path === "/chat")) return askClaude(request, env, cors); return reply({ error: "Unknown request" }, 404, cors); } }; // --------------------------------------------------------------------- // Claude: same as before // --------------------------------------------------------------------- async function askClaude(request, env, cors) { const text = await request.text(); if (text.length > 40000) return reply({ error: "Request too large" }, 413, cors); let body; try { body = JSON.parse(text); } catch (e) { return reply({ error: "Bad request" }, 400, cors); } const res = await fetch("https://api.anthropic.com/v1/messages", { method: "POST", headers: { "content-type": "application/json", "x-api-key": env.ANTHROPIC_API_KEY, "anthropic-version": "2023-06-01" }, body: JSON.stringify({ model: "claude-haiku-5-5", // fixed here so nobody can switch to a pricier model max_tokens: 500, // fixed cap on reply length (and cost) system: String(body.system || ""), messages: body.messages }) }); const data = await res.json().catch(() => ({})); if (!res.ok) return reply({ error: (data.error && data.error.message) || "Claude error " + res.status }, res.status, cors); const answer = (data.content || []).filter(b => b.type === "text").map(b => b.text).join("\n"); return reply({ reply: answer }, 200, cors); } // --------------------------------------------------------------------- // Live data: Google + fixtures, fetched at the same time. // If one fails, the other still comes back, plus a note saying what failed. // --------------------------------------------------------------------- async function getLive(env) { const [google, fixtures] = await Promise.allSettled([getGoogle(env), getFixtures()]); return { fetchedAt: new Date().toISOString(), places: google.status === "fulfilled" ? google.value : [], fixtures: fixtures.status === "fulfilled" ? fixtures.value : [], errors: [ google.status === "rejected" ? "Google: " + google.reason.message : null, fixtures.status === "rejected" ? "Fixtures: " + fixtures.reason.message : null ].filter(Boolean) }; } // One Google "Text Search" call returns up to 20 pubs inside the West End box async function getGoogle(env) { const res = await fetch("https://places.googleapis.com/v1/places:searchText", { method: "POST", headers: { "Content-Type": "application/json", "X-Goog-Api-Key": env.GOOGLE_PLACES_KEY, "X-Goog-FieldMask": GOOGLE_FIELDS }, body: JSON.stringify({ textQuery: "pub", locationRestriction: { rectangle: WEST_END_BOX }, pageSize: 20 }) }); const data = await res.json().catch(() => ({})); if (!res.ok) throw new Error((data.error && data.error.message) || "HTTP " + res.status); // Tidy each place into a short, flat object return (data.places || []).map(p => ({ name: p.displayName && p.displayName.text, lat: p.location && p.location.latitude, lon: p.location && p.location.longitude, rating: p.rating, reviews: p.userRatingCount, openNow: p.currentOpeningHours ? p.currentOpeningHours.openNow : undefined, dogs: p.allowsDogs, liveMusic: p.liveMusic, sport: p.goodForWatchingSports, food: (p.servesLunch || p.servesDinner) ? true : undefined, mapsUrl: p.googleMapsUri })); } // Today's fixtures from TheSportsDB (free public key "123") async function getFixtures() { const today = new Date().toISOString().slice(0, 10); // e.g. "2026-11-20" const base = "https://www.thesportsdb.com/api/v1/json/123/eventsday.php?d=" + today; const urls = Object.keys(LEAGUES).map(id => base + "&l=" + id); urls.push(base + "&s=Rugby"); const results = await Promise.allSettled(urls.map(u => fetch(u).then(r => r.json()))); const events = []; for (const r of results) { if (r.status !== "fulfilled" || !r.value || !r.value.events) continue; for (const e of r.value.events) { if (e.strSport === "Rugby" && !RUGBY_TEAMS.test(e.strEvent || "")) continue; events.push({ sport: e.strSport, league: e.strLeague, match: e.strEvent, startsUTC: e.strTimestamp || (e.dateEvent + "T" + (e.strTime || "00:00:00")) }); } } return events; } // Wraps an object up as a JSON response function reply(obj, status, headers) { return new Response(obj === null ? null : JSON.stringify(obj), { status, headers: { ...headers, "Content-Type": "application/json" } }); }
